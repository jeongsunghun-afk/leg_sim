cmake_minimum_required(VERSION 3.16)
project(quad_mpc_wbic_cpp CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")

# 의존성: proxddp conda env (mujoco C · eigen · eiquadprog)
set(ENV_PREFIX "/home/jsh/miniforge3/envs/proxddp" CACHE PATH "conda env prefix")

find_path(MUJOCO_INCLUDE  mujoco/mujoco.h              PATHS ${ENV_PREFIX}/include REQUIRED)
find_library(MUJOCO_LIB   mujoco                        PATHS ${ENV_PREFIX}/lib     REQUIRED)
find_path(EIGEN_INCLUDE   Eigen/Dense                   PATHS ${ENV_PREFIX}/include/eigen3 /usr/include/eigen3 REQUIRED)
find_path(EIQP_INCLUDE    eiquadprog/eiquadprog-fast.hpp PATHS ${ENV_PREFIX}/include REQUIRED)
find_library(EIQP_LIB     eiquadprog                    PATHS ${ENV_PREFIX}/lib     REQUIRED)

message(STATUS "MuJoCo: ${MUJOCO_LIB}")
message(STATUS "eiquadprog: ${EIQP_LIB}")

foreach(TGT quad_cpp trot_sim)
  if(TGT STREQUAL quad_cpp)
    add_executable(quad_cpp src/main.cpp)
  else()
    add_executable(trot_sim src/trot_sim.cpp)
  endif()
  target_include_directories(${TGT} PRIVATE ${MUJOCO_INCLUDE} ${EIGEN_INCLUDE} ${EIQP_INCLUDE})
  # ★conda MuJoCo가 최신 libstdc++(GLIBCXX_3.4.32)로 빌드됨 → conda libstdc++ 명시 링크(시스템 g++ 11.4 ABI 불일치 해결)
  target_link_libraries(${TGT} PRIVATE ${ENV_PREFIX}/lib/libstdc++.so ${MUJOCO_LIB} ${EIQP_LIB})
  set_target_properties(${TGT} PROPERTIES BUILD_RPATH "${ENV_PREFIX}/lib" INSTALL_RPATH "${ENV_PREFIX}/lib")
  target_link_options(${TGT} PRIVATE "-Wl,-rpath-link,${ENV_PREFIX}/lib")
endforeach()

# ── trot_view: GLFW 뷰어 (선택, GLFW 있을 때만) ──
find_library(GLFW_LIB glfw PATHS ${ENV_PREFIX}/lib)
find_path(GLFW_INCLUDE GLFW/glfw3.h PATHS ${ENV_PREFIX}/include /usr/include)
if(GLFW_LIB AND GLFW_INCLUDE)
  message(STATUS "GLFW: ${GLFW_LIB} → trot_view 빌드")
  add_executable(trot_view src/trot_view.cpp)
  target_include_directories(trot_view PRIVATE ${MUJOCO_INCLUDE} ${EIGEN_INCLUDE} ${EIQP_INCLUDE} ${GLFW_INCLUDE})
  target_link_libraries(trot_view PRIVATE ${ENV_PREFIX}/lib/libstdc++.so ${MUJOCO_LIB} ${EIQP_LIB} ${GLFW_LIB})
  set_target_properties(trot_view PROPERTIES BUILD_RPATH "${ENV_PREFIX}/lib" INSTALL_RPATH "${ENV_PREFIX}/lib")
  target_link_options(trot_view PRIVATE "-Wl,-rpath-link,${ENV_PREFIX}/lib")
else()
  message(STATUS "GLFW 없음 → trot_view 스킵")
endif()

# ── OCP: crocoddyl C++ 실시간 OCP (선택, crocoddyl 있을 때만) ──
#   §9 실시간 OCP 개발. jump OCP를 Python(offline replay)→C++ live-solve로 전환하는 스택.
#   ★find_package로 imported 타겟(crocoddyl::crocoddyl) 사용 = BOOST_MPL_LIMIT define·의존성(pinocchio·hpp-fcl 등) 자동.
#   (수동 -lpinocchio는 JointModel variant 25타입>boost.mpl한계20 컴파일 실패 → 반드시 imported 타겟)
list(APPEND CMAKE_PREFIX_PATH "${ENV_PREFIX}")
find_package(pinocchio QUIET)
find_package(crocoddyl QUIET)
if(pinocchio_FOUND AND crocoddyl_FOUND)
  message(STATUS "crocoddyl+pinocchio(find_package) → ocp 타겟 빌드")
  add_executable(ocp_check ocp/build_check.cpp)
  add_executable(jump_ocp  ocp/jump_ocp.cpp)
  foreach(OTGT ocp_check jump_ocp)
    target_link_libraries(${OTGT} PRIVATE crocoddyl::crocoddyl pinocchio::pinocchio ${ENV_PREFIX}/lib/libstdc++.so)
    set_target_properties(${OTGT} PROPERTIES BUILD_RPATH "${ENV_PREFIX}/lib" INSTALL_RPATH "${ENV_PREFIX}/lib")
    target_link_options(${OTGT} PRIVATE "-Wl,-rpath-link,${ENV_PREFIX}/lib")
  endforeach()
  # jump_ocp는 crouch IK(mj_crouch)를 MuJoCo로 self-contained 계산 → MuJoCo 링크
  target_include_directories(jump_ocp PRIVATE ${MUJOCO_INCLUDE})
  target_link_libraries(jump_ocp PRIVATE ${MUJOCO_LIB})
  # ★S3-b Step2: trot_view에도 crocoddyl 링크 → 점프모드 crouch중 live-solve(HAVE_CROCODDYL).
  #   (crocoddyl 없으면 trot_view는 그대로 빌드되고 점프는 /tmp/jump_traj.txt replay fallback)
  if(TARGET trot_view)
    message(STATUS "trot_view + crocoddyl → 점프 live-solve 활성(HAVE_CROCODDYL)")
    target_compile_definitions(trot_view PRIVATE HAVE_CROCODDYL)
    target_include_directories(trot_view PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/ocp)
    target_link_libraries(trot_view PRIVATE crocoddyl::crocoddyl pinocchio::pinocchio)
  endif()
else()
  message(STATUS "crocoddyl/pinocchio(find_package) 없음 → ocp 타겟 스킵")
endif()
